```python
from fastapi import FastAPI, Depends, BackgroundTasks
from redis.asyncio import Redis
import asyncpg

app = FastAPI(title="CoreEngine-API", version="3.0.0")

@app.get("/api/v1/telemetry/{service_id}")
async def get_service_telemetry(
    service_id: str,
    cache: Redis = Depends(get_cache_pool),
    db: asyncpg.Pool = Depends(get_db_pool)
):
    # Sub-5ms in-memory cache lookup
    cached = await cache.get(f"telemetry:{service_id}")
    if cached:
        return {"source": "cache", "payload": cached}

    # Atomic query execution with indexed reads
    record = await db.fetchrow(
        "SELECT * FROM telemetry WHERE id = $1", service_id
    )
    await cache.setex(f"telemetry:{service_id}", 60, record)
    return {"source": "postgres", "payload": record}
```
